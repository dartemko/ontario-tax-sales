Automated Rescraper Pipeline

Python BeautifulSoup script and GitHub Actions cron workflow for auto-updating public tax tender listings.

1. Python Gazette Scraper (`scraper.py`)

import requests
from bs4 import BeautifulSoup
import re
import json

URL = "https://www.ontario.ca/document/ontario-gazette"
HEADERS = {"User-Agent": "Mozilla/5.0"}

def fetch_tax_sales():
    resp = requests.get(URL, headers=HEADERS)
    soup = BeautifulSoup(resp.content, 'html.parser')
    listings = []
    
    for block in soup.find_all(['div', 'section']):
        text = block.get_text()
        if "SALE OF LAND BY PUBLIC TENDER" in text:
            pin = re.search(r'PIN\s*#?\s*([\d-]+)', text)
            bid = re.search(r'Minimum\s+Tender\s+Amount:\s*\$([\d,]+\.\d{2})', text)
            if bid:
                listings.append({
                    "pin": pin.group(1) if pin else "N/A",
                    "minBid": float(bid.group(1).replace(',', '')),
                    "noticeUrl": URL,
                    "tenderFormPdf": "https://forms.mgcs.gov.on.ca/dataset/402ef9c5-2da6-4880-98be-e2e897627bcd/resource/31f730ca-c641-4645-8473-9ddd90bea745/download/2205e.pdf"
                })
    return listings

if __name__ == "__main__":
    with open("listings.json", "w") as f:
        json.dump(fetch_tax_sales(), f, indent=2)

2. GitHub Actions Workflow (`.github/workflows/scraper.yml`)

name: Auto Rescrape Ontario Tax Sales

on:
  schedule:
    - cron: '0 0 * * 0' # Auto-runs midnight every Sunday
  workflow_dispatch:

jobs:
  scrape:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - uses: actions/setup-python@v4
        with:
          python-version: '3.10'
      - run: pip install requests beautifulsoup4
      - run: python scraper.py
      - run: |
          git config --global user.name 'github-actions'
          git config --global user.email 'actions@github.com'
          git add listings.json
          git commit -m "Auto-update tax sale listings" || exit 0
          git push
